\section{Preliminaries}\label{s:preliminaries}

Let $r$ be a positive integer, let $\Cyc_r = \langle \rho\mid \rho^r\rangle$ be the cyclic group of order $r$ and let $R$ be a commutative ring with unit. Let $\Mod{R}$ be the category of $R$-modules.




\subsection{Chain complexes and sign conventions}

A \emph{chain complex} is a graded $R$-module $C$ with a graded endomorphism $\partial \colon C\to C$ of degree $-1$ such that $\partial\circ \partial =0$. Let $\Ch{R}$ be the category of chain complexes over $R$ and chain maps between them.


\subsubsection{Tensor products}	If $A$ and $B$ are chain complexes whose differentials have the same degree, their tensor product $A \ot B$ is the graded tensor product of $A$ and $B$ with differential $\partial(a \ot b) = \partial a \ot b + (-1)^{|a|}a \ot \partial b$. If $f\colon A\to B$ and $g\colon C\to D$ are chain maps, then $f\ot g\colon A\ot C\to B\ot D$ is defined as $(f\ot g)(a\ot c) = (-1)^{|a||g|}f(a)\ot g(c)$.

If $(C,\partial)$ is a chain complex, its left suspension $\susp{}C$ is the chain complex $(\susp{}C)_k = C_{k-1}$ with differential $-\partial$. Its right suspension $\sus{}C$ is the chain complex $\sus{}C_k = C_{k-1}$ with differential $\partial$. If $R[1]$ denotes the chain complex that consists of a single copy of $R$ in degree $1$, there are isomorphisms 
\begin{align}\label{eq:401}
	\sus{} \ucadenas &\cong \ucadenas \ot R[1] & \susp{} \ucadenas \cong R[1] \ot \ucadenas.
\end{align}
that send $a$ to $a \ot 1$ and to $1 \ot a$ respectively. From this viewpoint left and right suspensions are functors that act on a chain map $f$ as $(\sus{}f) = f\ot \id$ and $(\susp{} f) = \id\ot f$.


\subsubsection{Linear dual} The linear dual of a complex $\ucadenas$ is the chain complex $\ucocadenas = \hom_R(\ucadenas,R)$, where $R$ is concentrated in degree $0$. Thus, if $\ucadenas$ is positively graded, then $\ucocadenas$ is negatively graded. The differential in the dual complex $(\ucocadenas,\partial^\dd)$ is
\begin{align*}
	(\partial^\dd f)(a) &= (-1)^{|f|+1} f(\partial a). 
\end{align*} 

Most of the chain complexes in this work will be finitely generated free complexes with a prescribed basis. We will denote the dual of a basis element $u$ by $u^\dd$. 


The linear dual of a chain map $f\colon \ucadenas \to D$ will be denoted $f^{\dd}\colon D^\dd\to \ucocadenas$.  
There is an isomorphism $A^\dd \ot B^\dd \to (B \ot A)^{\dd}$ that sends $f \ot g$ to $h$ where $h(b \ot a) = g(b)\cdot f(a)$. As a consequence of this and \eqref{eq:401}, there is an isomorphism
\begin{align*}
	(\sus{}\ucadenas )^\dd &\cong \susp{-1}\ucocadenas
\end{align*}
that sends a cochain $f$ to the cochain $\susp{}f$ defined as $(\susp{}f)(a) = f(a\ot 1)$.

\subsubsection{Cyclic actions}

Let $\ucadenas$ be a chain complex of left $R[\Cyc_r]$-modules. The linear dual $\ucocadenas=\hom_{R[\Cyc_r]}(C, R[\Cyc_r])$ is a chain complex of right $R[\Cyc_r]$-modules with right $\Cyc_r$-action given by $ (f\cdot \rho)(a)=f(a)\rho$. Thus, if $u^\dd$ is the dual of a basis element of $\ucadenas$, the cyclic group acts backwards, in the sense that $u^\dd\cdot \rho=(\rho^{-1}u)^\dd$.

Via the group anti-automorphism $\rho\mapsto\rho^{-1}$, we can endow $\ucocadenas$ with a left $\Cyc_r$-action, defined as 
\[
\rho\cdot u^\dd:=u^\dd\cdot \rho^{-1}=(\rho u)^\dd.
\] Note that $(\rho\cdot f)(a)=f(a)\rho^{-1}$, not $\rho f(a)$. We will always consider these left cyclic actions for the dual chain complexes.

\subsection{Cyclic coalgebras}\label{s:categories} Here we define $\diag_r$-coalgebras and give three examples. Note that we do not impose any axiom (associativity, unitality).

\begin{definition}
	A $\diag_r$-coalgebra is an $R$-module $M$ together with a homomorphism $\diag_r\colon M\to M^{\ot r}$ equivariant with respect to the action of $\Cyc_r$ (which is trivial on the source). The homomorphism $\diag$ is called \emph{cyclic $r$-diagonal}. A map of $\diag_r$-coalgebras is a homomorphism of $R$-modules that commutes with the cyclic $r$-diagonals.  
\end{definition}
\begin{example}
	The category of free $R$-modules with a prescribed basis, and with morphisms those homomorphisms that send generators to generators (we call these homomorphisms \emph{free}). This is the image of the free functor $\Set\to \Mod{R}$. If $X$ is a set, the diagonal $X\to X^{\times r}$ induces a cyclic $r$-diagonal $\diag_r\colon R\langle X\rangle \to R\langle X\rangle^{\ot r}$ on this category.
\end{example}
\begin{example}
		The category of free $R$-modules with a prescribed basis with morphisms those homomorphisms that send generators to generators or to zero (we call these homomorphisms \emph{almost free}). This is the image of the free functor $\Setp\to \Mod{R}$ that sends a pointed set $X$ to the quotient $R\langle (X,\{pt\})\rangle$ of $R\langle X\rangle$ by the submodule generated by the basepoint. If $X$ is a pointed set, the diagonal $X\to X^{\wedge r}$ induces a cyclic $r$-diagonal $\diag_r\colon R\langle (X,\{pt\})\rangle \to R\langle (X,\{pt\})\rangle^{\ot r}$ on this category.
\end{example}

\begin{example}
	The category of free $R$-modules with a prescribed basis with morphisms those homomorphisms that send generators to generators (possibly with a sign) or to zero (we call these homomorphisms almost free signed). The cyclic $r$-diagonal $\diag_r\colon R\langle (X,\{pt\})\rangle \to R\langle (X,\{pt\})\rangle^{\ot r}$ is defined as $\diag_r(x) = x\ot\ldots \ot x$. This map is natural only if $r$ is odd.
\end{example}

\begin{example}
	If $p$ is prime, the category of free $\bF_p$-vector spaces with a prescribed basis with morphisms those homomorphisms that send generators to scalar multiples of generators. The cyclic $r$-diagonal $\diag_r\colon R\langle (X,\{pt\})\rangle \to R\langle (X,\{pt\})\rangle^{\ot r}$ is defined as $\diag_r(x) = x\ot\ldots \ot x$, and only for $r=p$.
\end{example}

\subsection{Augmented simplicial objects}\label{s:simplices}

For each integer $n \geq 0$, denote the $n$th ordinal by $[n] = \{0,\dots,n-1\}$ , with $[0]$ denoting the empty ordinal. Let $\cC$ be a cocomplete category with a strict monoidal tensor product (in practice, it will be either $\Mod{R}$, $\Ch{R}$ or a category of $\diag_p$-coalgebras).


\begin{warning}
	This is not the standard notation, but is very convenient for the join product.
\end{warning}

\begin{definition}
	The \emph{augmented simplex category} $\asimplex$ is the category of finite ordinals and order-preserving maps between them. The \emph{simplex category} $\simplex$ is the category of finite non-empty ordinals and order-preserving maps between them.  
	The \emph{augmented semi-simplex category} $\asimplexinj$ is the category of finite ordinals and order-preserving injections between them. The \emph{semi-simplex category} $\simplexinj$ is the category of finite non-empty ordinals and injective order-preserving maps between them. We will write $\nabla$ for any of these four categories.
\end{definition}



The $i$th coface is the injective order-preserving map $\d^i\colon [n] \to [n+1]$ that misses $i$ in the target. The i$th$ codegeneracy is the surjective order-preserving map $\s^i\colon [n]\to [n+1]$ that sends $i$ and $i+1$ to the same element. Every map in the (augmented) simplex category can be decomposed as a sequence of codegeneracies followed by a sequence of cofaces, and every map in the (augmented) semi-simplex category is a composition of cofaces.


\begin{definition}
	An \emph{augmented simplicial object in $\cC$} is a functor $X \colon \asimplex^\mathrm{op} \to \cC$.
	An \emph{augmented cosimplicial object in $\cC$} is a functor $X \colon \asimplex \to \cC$. We have similar definitions for the other three simplex categories.
\end{definition}

If $X$ is a $\nabla^{\op}$-object, $X([n])$ is denoted $X_n$ and the morphisms $X(d^i)$ and $X(s^i)$ are denoted $\d_i$ and $\s_i$ respectively. If $X$ is a $\nabla$-object, $X([n])$ is denoted $X_n$ and the morphisms $X(d^i)$ and $X(s^i)$ are denoted $\d^i$ and $\s^i$ respectively.
We will often write $X_\bullet$ to stress that $X$ is a $\nabla^{\op}$-object or a $\nabla$-object.

The inclusion $\simplex \subset \asimplex$ of the non-empty ordinals induces a restriction functor $\cC^{\asimplex^{\op}} \to \cC^{\simplex^{\op}}$ to the category of simplicial objects on $\cC$ that forgets $X_{0}$. It has a left adjoint that sets $X_{0}$ to be the colimit of $X$ and a right adjoint that sets $X_{0}$ to be the terminal object of $\cC$.

There are functors $\cC^{\simplex^{\op}}\to \cC^{\simplexinj^{\op}}$ and $\cC^{\asimplex^{\op}}\to \cC^{\asimplexinj^{\op}}$ that send an (augmented) simplicial object $X_\bullet$ to the (augmented) semi-simplicial object $X_\bullet^\nd$ whose $n$-simplices are the quotient of the $n$-simplices of $X_\bullet$ by the image of the degeneracies, and whose face maps are the maps induced on the quotient by the face maps of $X_\bullet$.


%If $X$ is an augmented simplicial object in one of the categories of Section \ref{s:categories}, postcomposing with the linearization functor $\L \colon \cC \to \Mod{R}$ yields an augmented simplicial $R$-module $\abel(X)$.

The complex of normalised chains $\chains(X)$ of an (augmented) semi-simplicial $R$-module is defined as
\begin{align*}
	\chains(X)_n &= X_n
	&
	\partial(x) &= \sum_{j=0}^{n-1} (-1)^j \d_j(x)
\end{align*}
The \emph{complex of normalized chains} $\chains(X)$ of an (augmented) simplicial $R$-module is the complex $\chains(X^{\mathrm{nd}})$. If $X$ is a augmented semi-simplicial chain complex, then $\chains(X)$ is a bicomplex, and we will abuse notation calling $\chains(X)$ the totalization of this bicomplex.


\begin{warning}
	Because of our unusual but convenient convention to denote ordinals, the `vertices'' of a simplicial $R$-module are $1$-simplices (instead of $0$-simplices) and the augmentation lives in dimension $0$ (instead of dimension $-1$). The chain complex $\chains(X)$ is the right suspension of the usual chain complex of $X$.
\end{warning}



The \emph{standard augmented $n$-simplex} is the augmented simplicial $R$-module $\simplex^n_+$ given by by $[m] \to R\mathrm{Mor}_{\asimplex}([m],[n])$. The normalized chain complex $\chains(\asimplex^n)$ is generated by the inclusions $[m] \to [n]$, which we denote by their images. If $\perm$ is a permutation of $0,\dots,m-1$, then $[v_{\perm(0)},\dots,v_{\perm(m-1)}]$ will denote the generator $(-1)^{\sign{\perm}}[v_0,\dots,v_{m-1}]$.

\begin{warning}
	 Note that with our convention $\simplex^n$ has $n$ vertices instead of $n+1$.
\end{warning}

If $X,Y$ are $\nabla$-objects, their \emph{tensor product} $X\ot Y$ is the $\nabla$-object with $(X\ot Y)_n = X_n\ot Y_n$ and $\d_i(x\ot y) = \d_i(x)\ot \d_i(y)$ and $\s_i(x\ot y) = \s_i(x)\ot \s_i(y)$. In general $\chains(X\ot Y)\not\simeq \chains(X)\ot \chains(Y)$, only when $\nabla$ is the simplex category $\simplex$ the Eilenberg-Zilber theorem gives a homotopy equivalence $\chains(X\ot Y)\simeq \chains(X)\ot \chains(Y)$.


Each simplex category $\nabla$ has a strict monoidal product called the \emph{join}, that sends a pair of ordinals $[n],[m]$ to the ordinal $[n+m]$ and a pair of functions $f \colon [n] \to [q]$, $g \colon [m] \to [p]$ to the function $f*g \colon [n+m] \to [q+p]$ defined as $(f*g)(x) = f(x)$ if $x< n$ and $(f*g)(x) = q+g(x-n)$ if $x \geq n$. The unit is the empty ordinal $[0]$. This product induces a strict monoidal product via Day convolution on $\nabla$-objects, that we denote as $X*_\nabla Y$ (the \emph{join product}). We have that $\chains(X*_\nabla Y)\cong \chains(X)\ot \chains(Y)$.
